def create_tasks():

    research_task = Task(
        description=(
            "Analyze the supplied YouTube transcript carefully.\n\n"

            "VIDEO TITLE: {video_title}\n"
            "VIDEO URL: {video_url}\n\n"

            "TRANSCRIPT:\n"
            "{transcript}\n\n"

            "YOUR RESPONSIBILITY:\n"
            "Extract reliable research notes for a short technical blog.\n\n"

            "Identify:\n"
            "1. The central subject of the video.\n"
            "2. The most important concepts explained.\n"
            "3. Technical terminology and its meaning.\n"
            "4. Practical examples explicitly discussed.\n"
            "5. Important relationships between concepts.\n\n"

            "STRICT ACCURACY RULES:\n"
            "- Use only information supported by the transcript.\n"
            "- Never invent examples, statistics, applications or facts.\n"
            "- Do not add outside knowledge to fill missing information.\n"
            "- If an example is unclear or absent, leave it out.\n"
            "- Do not mistake speculation for something stated in the video.\n"
            "- Do not repeat the same concept in different words.\n\n"

            "OUTPUT RULES:\n"
            "- Do not write the blog yourself.\n"
            "- Produce structured, concise research notes.\n"
            "- Maximum 150 words.\n"
            "- Include only information useful for the final blog."
        ),
        expected_output=(
            "Accurate research notes of no more than 150 words, "
            "covering the video's central topic, key concepts, "
            "technical explanations and transcript-supported examples."
        ),
        agent=blog_researcher
    )

    write_task = Task(
        description=(
            "Write a concise and technically accurate blog using "
            "ONLY the research notes produced by the researcher.\n\n"

            "VIDEO TITLE: {video_title}\n\n"

            "STRICT CONTENT REQUIREMENTS:\n"
            "1. Write approximately 300-400 tokens in total, "
            "including the title.\n"
            "2. Start with one meaningful Markdown heading.\n"
            "3. Introduce the central topic briefly.\n"
            "4. Explain the most important concepts clearly.\n"
            "5. Include one transcript-supported example if available.\n"
            "6. End with a short conclusion.\n"
            "7. Use clear, professional and technically accurate language.\n\n"

            "STRICT ACCURACY REQUIREMENTS:\n"

        context=[research_task],
        async_execution=False

import time

import tiktoken
from dotenv import load_dotenv
from crewai import Agent, Crew, Process, Task
from langsmith import traceable

load_dotenv()


# --------------------------------------------------
# TOKEN COUNTING
# --------------------------------------------------

def count_tokens(text: str) -> int:
    encoding = tiktoken.get_encoding("cl100k_base")
    return len(encoding.encode(text))


# --------------------------------------------------
# 1. RESEARCHER AGENT EXECUTION
# --------------------------------------------------

@traceable(
    name="researcher_agent",
    run_type="chain",
    process_inputs=lambda inputs: {
        **inputs,
        "transcript": (
            f"[Transcript redacted: "
            f"{len(inputs.get('transcript', ''))} characters]"
        )
    }
)
def run_researcher(video_url: str, video_title: str, transcript: str):

    from agents import blog_researcher
    from tasks import create_tasks

    research_task, _ = create_tasks()

    research_crew = Crew(
        agents=[blog_researcher],
        tasks=[research_task],
        process=Process.sequential,
        memory=False,
        cache=True,
        verbose=True
    )

    result = research_crew.kickoff(
        inputs={
            "video_url": video_url,
            "video_title": video_title,
            "transcript": transcript
        }
    )

    research_notes = result.raw.strip()

    if not research_notes:
        raise ValueError("Researcher returned empty research notes.")

    return research_notes


# --------------------------------------------------
# 2. WRITER AGENT EXECUTION
# --------------------------------------------------

@traceable(
    name="writer_agent",
    run_type="chain"
)
def run_writer(
    video_title: str,
    research_notes: str,
    writer_description: str,
    expected_output: str
):

    from agents import blog_writer

    description = writer_description.replace(
        "{video_title}",
        video_title
    )

    description += (
        "\n\nRESEARCH NOTES FROM THE RESEARCHER:\n"
        f"{research_notes}\n\n"
        "Use only these research notes as factual source material."
    )

    writer_task = Task(
        description=description,
        expected_output=expected_output,
        agent=blog_writer,
        async_execution=False
    )

    writer_crew = Crew(
        agents=[blog_writer],
        tasks=[writer_task],
        process=Process.sequential,
        memory=False,
        cache=False,
        verbose=True
    )

    result = writer_crew.kickoff()

    blog = result.raw.strip()

    if not blog:
        raise ValueError("Writer returned an empty blog.")

    token_count = count_tokens(blog)

    print(f"\nInitial blog token count: {token_count}")

    # One revision attempt if the blog exceeds 400 tokens.
    if token_count > 400:

        print("Blog exceeds 400 tokens. Requesting revision...")

        revision_task = Task(
            description=(
                "Rewrite this blog to contain 300-400 tokens, "
                "including its title.\n\n"
                "Preserve the important technical information.\n"
                "Use only facts from the original blog.\n"
                "Do not introduce new facts or examples.\n"
                "Return only the revised Markdown blog.\n\n"
                f"ORIGINAL BLOG:\n{blog}"
            ),
            expected_output=(
                "A concise Markdown blog of 300-400 tokens."
            ),
            agent=blog_writer,
            async_execution=False
        )

        revision_crew = Crew(
            agents=[blog_writer],
            tasks=[revision_task],
            process=Process.sequential,
            memory=False,
            cache=False,
            verbose=True
        )

        revised_result = revision_crew.kickoff()
        revised_blog = revised_result.raw.strip()
        revised_count = count_tokens(revised_blog)

        print(f"Revised blog token count: {revised_count}")

        if abs(revised_count - 350) < abs(token_count - 350):
            blog = revised_blog
            token_count = revised_count

    print(f"Final blog token count: {token_count}")

    return {
        "blog": blog,
        "token_count": token_count
    }


# --------------------------------------------------
# 3. FINAL WORKFLOW OUTPUT
# --------------------------------------------------

@traceable(
    name="final_workflow_output",
    run_type="chain"
)
def record_final_output(
    video_title: str,
    blog: str,
    token_count: int,
    execution_time: float
):

    return {
        "status": "success",
        "video_title": video_title,
        "blog": blog,
        "token_count": token_count,
        "execution_time_seconds": round(execution_time, 2)
    }


# --------------------------------------------------
# PARENT WORKFLOW
# --------------------------------------------------

@traceable(
    name="youtube_blog_workflow",
    run_type="chain",
    process_inputs=lambda inputs: {
        **inputs,
        "transcript": (
            f"[Transcript redacted: "
            f"{len(inputs.get('transcript', ''))} characters]"
        )
    }
)
def generate_blog(
    video_url: str,
    video_title: str,
    transcript: str
):

    from tasks import create_tasks

    start_time = time.perf_counter()

    # Create the original task definitions.
    _, write_task = create_tasks()

    # Stage 1: Researcher.
    research_notes = run_researcher(
        video_url=video_url,
        video_title=video_title,
        transcript=transcript
    )

    # Stage 2: Writer.
    writer_result = run_writer(
        video_title=video_title,
        research_notes=research_notes,
        writer_description=write_task.description,
        expected_output=write_task.expected_output
    )

    blog = writer_result["blog"]
    token_count = writer_result["token_count"]

    execution_time = time.perf_counter() - start_time

    # Stage 3: Final output.
    final_result = record_final_output(
        video_title=video_title,
        blog=blog,
        token_count=token_count,
        execution_time=execution_time
    )

    print("\n========== WORKFLOW SUMMARY ==========")
    print("Status:", final_result["status"])
    print("Video:", video_title)
    print("Blog tokens:", token_count)
    print("Execution time:", final_result["execution_time_seconds"], "seconds")
    print("======================================")

    return final_result["blog"]

import re
import requests

from functools import lru_cache
from typing import Type
from urllib.parse import urlparse, parse_qs

from pydantic import BaseModel, Field
from crewai.tools import BaseTool
from youtube_transcript_api import YouTubeTranscriptApi


# --------------------------------------------------
# YOUTUBE URL VALIDATION
# --------------------------------------------------

def extract_video_id(video_url: str) -> str:

    if not video_url or not video_url.strip():
        raise ValueError("YouTube URL cannot be empty.")

    parsed = urlparse(video_url.strip())
    hostname = parsed.netloc.lower().split(":")[0]

    video_id = None

    if hostname in ("youtu.be", "www.youtu.be"):
        video_id = parsed.path.strip("/").split("/")[0]

    elif hostname in (
        "youtube.com",
        "www.youtube.com",
        "m.youtube.com",
        "music.youtube.com"
    ):
        if parsed.path == "/watch":
            video_id = parse_qs(parsed.query).get("v", [None])[0]

        else:
            match = re.match(
                r"^/(?:shorts|embed|live|v)/([^/?]+)",
                parsed.path
            )

            if match:
                video_id = match.group(1)

    if not video_id or not re.fullmatch(
        r"[A-Za-z0-9_-]{11}", video_id
    ):
        raise ValueError("Invalid or unsupported YouTube URL.")

    return video_id


# --------------------------------------------------
# VIDEO TITLE
# --------------------------------------------------

@lru_cache(maxsize=32)
def fetch_video_title(video_id: str) -> str:

    video_url = (
        f"https://www.youtube.com/watch?v={video_id}"
    )

    try:
        response = requests.get(
            "https://www.youtube.com/oembed",
            params={
                "url": video_url,
                "format": "json"
            },
            timeout=20
        )

        response.raise_for_status()

        data = response.json()

        return data.get("title", "YouTube Video")

    except requests.RequestException:
        return f"YouTube Video ({video_id})"


# --------------------------------------------------
# TRANSCRIPT EXTRACTION
# --------------------------------------------------

@lru_cache(maxsize=16)
def _fetch_transcript_data(video_id: str):

    try:
        api = YouTubeTranscriptApi()

        transcript_data = api.fetch(
            video_id,
            languages=["en"]
        )

        transcript = "\n".join(
            segment.text
            for segment in transcript_data
        )

        language = transcript_data.language

        if not transcript.strip():
            raise ValueError(
                "The video returned an empty transcript."
            )

        return transcript, language

    except Exception as error:
        raise RuntimeError(
            f"Unable to retrieve YouTube transcript: {error}"
        ) from error


def fetch_video_transcript(video_id: str):

    return _fetch_transcript_data(video_id)


# --------------------------------------------------
# CREWAI TOOL
# --------------------------------------------------

class YouTubeTranscriptInput(BaseModel):

    video_url: str = Field(
        description="Public YouTube video URL."
    )


class YouTubeTranscriptTool(BaseTool):

    name: str = "YouTube Transcript Extractor"

    description: str = (
        "Retrieves available English captions directly from "
        "YouTube using the youtube-transcript-api package. "
        "Does not require an API key or local transcription model."
    )

    args_schema: Type[BaseModel] = YouTubeTranscriptInput

    def _run(self, video_url: str) -> str:

        try:
            video_id = extract_video_id(video_url)

            title = fetch_video_title(video_id)

            transcript, language = fetch_video_transcript(
                video_id
            )

            return (
                f"VIDEO TITLE: {title}\n"
                f"LANGUAGE: {language}\n"
                f"COMPLETE TRANSCRIPT:\n\n"
                f"{transcript}"
            )

        except Exception as error:
            return f"Transcript extraction error: {error}"
